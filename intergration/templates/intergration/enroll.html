Add a Wazuh Agent

Install the Wazuh agent on your endpoint using the commands below. It will automatically register to your tenant group.

Your Group
{{ enroll.group }}
Linux / Ubuntu / Debian
curl -s https://packages.wazuh.com/key/GPG-KEY-WAZUH | gpg --no-default-keyring \
  --keyring gnupg-ring:/usr/share/keyrings/wazuh.gpg --import && \
  chmod 644 /usr/share/keyrings/wazuh.gpg
echo "deb [signed-by=/usr/share/keyrings/wazuh.gpg] https://packages.wazuh.com/4.x/apt/ stable main" | \
  tee /etc/apt/sources.list.d/wazuh.list
apt-get update
WAZUH_MANAGER="{{ enroll.wazuh_host }}" \
WAZUH_REGISTRATION_SERVER="{{ enroll.wazuh_host }}" \
WAZUH_AGENT_GROUP="{{ enroll.group }}" \
apt-get install -y wazuh-agent
systemctl enable wazuh-agent && systemctl start wazuh-agent
Windows (PowerShell — run as Administrator)
Invoke-WebRequest -Uri https://packages.wazuh.com/4.x/windows/wazuh-agent-4.x.x-1.msi `
  -OutFile wazuh-agent.msi
msiexec /i wazuh-agent.msi /q `
  WAZUH_MANAGER="{{ enroll.wazuh_host }}" `
  WAZUH_REGISTRATION_SERVER="{{ enroll.wazuh_host }}" `
  WAZUH_AGENT_GROUP="{{ enroll.group }}"
NET START WazuhSvc
macOS
curl -so wazuh-agent.pkg https://packages.wazuh.com/4.x/macos/wazuh-agent-4.x.x-1.pkg
echo "WAZUH_MANAGER='{{ enroll.wazuh_host }}' \
WAZUH_REGISTRATION_SERVER='{{ enroll.wazuh_host }}' \
WAZUH_AGENT_GROUP='{{ enroll.group }}'" > /tmp/wazuh_envs
sudo installer -pkg wazuh-agent.pkg -target /
sudo /Library/Ossec/bin/wazuh-control start
